22inline bool coversDisplay(const Bounds &window, const Bounds &display, double edgeTolerance = 8.0)
23{
24 const auto displayWidth = display.right - display.left;
25 const auto displayHeight = display.bottom - display.top;
26 if (displayWidth <= 0.0 || displayHeight <= 0.0) {
27 return false;
28 }
29
30 const auto intersectionWidth =
31 (std::max)(0.0, (std::min)(window.right, display.right) - (std::max)(window.left, display.left));
32 const auto intersectionHeight =
33 (std::max)(0.0, (std::min)(window.bottom, display.bottom) - (std::max)(window.top, display.top));
34 const auto coverage = intersectionWidth * intersectionHeight / (displayWidth * displayHeight);
35
36 return std::isfinite(coverage) && coverage >= 0.98 && window.left <= display.left + edgeTolerance &&
37 window.top <= display.top + edgeTolerance && window.right >= display.right - edgeTolerance &&
38 window.bottom >= display.bottom - edgeTolerance;
39}

41inline bool
42pointerIsConfinedToDisplay(const Bounds &clip, const Bounds &display, const Bounds &desktop, double edgeTolerance = 4.0)
43{
44 const auto clipWidth = clip.right - clip.left;
45 const auto clipHeight = clip.bottom - clip.top;
46 if (clipWidth < 64.0 || clipHeight < 64.0) {
47 return false;
48 }
49
50 // The ordinary unclipped state covers the complete virtual desktop. A
51 // smaller rectangle contained by one physical display is an explicit
52 // pointer-capture request, which games commonly use in both exclusive and
53 // borderless/windowed modes.
54 if (coversDisplay(clip, desktop, edgeTolerance)) {
55 return false;
56 }
57
58 return clip.left >= display.left - edgeTolerance && clip.top >= display.top - edgeTolerance &&
59 clip.right <= display.right + edgeTolerance && clip.bottom <= display.bottom + edgeTolerance;
60}
