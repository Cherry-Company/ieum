28inline bool reconcileMacLoginItems(
29 bool enabled, const MacLoginItemOperations &mainApp, const MacLoginItemOperations &legacyAgent, QString *error
30)
31{
32 const auto status = mainApp.status();
33 const bool alreadyRegistered =
34 status == MacLoginItemStatus::Enabled || status == MacLoginItemStatus::RequiresApproval;
35 const bool changeMainApp = enabled ? !alreadyRegistered : status != MacLoginItemStatus::NotRegistered;
36 QString mainError;
37 const bool mainOk = !changeMainApp || mainApp.setEnabled(enabled, &mainError);
38
39 // Register the replacement first: unregistering an agent can terminate its
40 // running process. Always retire the old job, even when the main app already
41 // has the desired status or registration failed. Otherwise it can keep
42 // producing launch-constraint crash reports on every login.
43 QString legacyError;
44 const bool legacyOk =
45 legacyAgent.status() == MacLoginItemStatus::NotRegistered || legacyAgent.setEnabled(false, &legacyError);
46 if (error != nullptr) {
47 *error = mainError;
48 if (!legacyOk) {
49 if (!error->isEmpty()) {
50 error->append(QLatin1Char('\n'));
51 }
52 error->append(QStringLiteral("Could not remove the previous Ieum login agent: %1").arg(legacyError));
53 }
54 }
55 return mainOk && legacyOk;
56}
