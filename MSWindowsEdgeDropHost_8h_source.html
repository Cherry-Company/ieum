53[[nodiscard]] inline std::vector<MSWindowsEdgeDropWindow> makeMSWindowsEdgeDropLayout(
54 const MSWindowsEdgeDropBounds &desktop, std::uint32_t activeSides, std::int32_t thickness = 3
55)
56{
57 if (desktop.width <= 0 || desktop.height <= 0 || thickness <= 0 || activeSides == 0) {
58 return {};
59 }
60
61 const auto right64 = static_cast<std::int64_t>(desktop.x) + desktop.width;
62 const auto bottom64 = static_cast<std::int64_t>(desktop.y) + desktop.height;
63 if (right64 > (std::numeric_limits<std::int32_t>::max)() || right64 < (std::numeric_limits<std::int32_t>::min)() ||
64 bottom64 > (std::numeric_limits<std::int32_t>::max)() || bottom64 < (std::numeric_limits<std::int32_t>::min)()) {
65 return {};
66 }
67
68 const auto edge = (std::min)({thickness, desktop.width, desktop.height});
69 const auto right = static_cast<std::int32_t>(right64);
70 const auto bottom = static_cast<std::int32_t>(bottom64);
71 std::vector<MSWindowsEdgeDropWindow> result;
72 result.reserve(4);
73
74 const auto hasSide = [activeSides](DirectionMask side) {
75 return (activeSides & static_cast<std::uint32_t>(side)) != 0;
76 };
77 using enum DirectionMask;
78 if (hasSide(LeftMask)) {
79 result.push_back({Direction::Left, desktop.x, desktop.y, edge, desktop.height});
80 }
81 if (hasSide(RightMask)) {
82 result.push_back({Direction::Right, right - edge, desktop.y, edge, desktop.height});
83 }
84
85 const auto horizontalWidth64 = static_cast<std::int64_t>(desktop.width) - (2LL * edge);
86 if (horizontalWidth64 > 0 && hasSide(TopMask)) {
87 result.push_back({Direction::Top, desktop.x + edge, desktop.y, static_cast<std::int32_t>(horizontalWidth64), edge});
88 }
89 if (horizontalWidth64 > 0 && hasSide(BottomMask)) {
90 result.push_back(
91 {Direction::Bottom, desktop.x + edge, bottom - edge, static_cast<std::int32_t>(horizontalWidth64), edge}
92 );
93 }
94
95 return result;
96}
