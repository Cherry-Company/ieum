16// Release follows the original press, even after a plan or target changes.
17class BroadcastKeys
18{
19public:
20 struct Press

22 KeyID key;

26 bool remember(KeyButton button, KeyID key, KeyModifierMask mask, std::vector<std::string> recipients)
27 {
28 if (recipients.empty())
29 return false;
30 return m_pressed.try_emplace(button, Press{key, mask, std::move(recipients)}).second;
31 }
32 const Press *find(KeyButton button) const
33 {
34 const auto it = m_pressed.find(button);
35 return it == m_pressed.end() ? nullptr : &it->second;
36 }
37 std::optional<Press> take(KeyButton button)
38 {
39 auto node = m_pressed.extract(button);
40 if (node.empty())
41 return std::nullopt;
42 return std::move(node.mapped());
43 }
44 auto takeAll()
45 {
46 return std::exchange(m_pressed, {});
47 }
48 auto releaseAll()
49 {
50 auto pressed = m_pressed;
51 for (auto &[button, press] : m_pressed)
52 press.recipients.clear();
53 // Consume the eventual physical key-up and repeats without forwarding
54 // them to a newly selected screen after a toggle or entitlement expiry.
55 return pressed;
56 }
57 void forgetPeer(const std::string &name)
58 {
59 for (auto &[button, press] : m_pressed)
60 std::erase(press.recipients, name);
61 // Keep empty records until key-up; reconnecting the same name must not
62 // inherit a key pressed in its previous connection.
63 }
64
65private:
66 std::map<KeyButton, Press> m_pressed;
67};
68} // namespace deskflow::server
