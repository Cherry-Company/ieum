25inline int32_t clampCoordinate(int32_t coordinate, int32_t origin, int32_t extent)
26{
27 if (extent <= 0) {
28 return origin;
29 }
30
31 const auto lower = static_cast<int64_t>(origin);
32 const auto upper =
33 std::min(lower + static_cast<int64_t>(extent) - 1, static_cast<int64_t>(std::numeric_limits<int32_t>::max()));
34 return static_cast<int32_t>(std::clamp(static_cast<int64_t>(coordinate), lower, upper));
35}

37inline float toFraction(int32_t coordinate, int32_t origin, int32_t extent)
38{
39 if (extent <= 0) {
40 return 0.5f;
41 }
42
43 const auto clamped = clampCoordinate(coordinate, origin, extent);
44 return static_cast<float>(
45 (static_cast<double>(clamped) - static_cast<double>(origin) + 0.5) / static_cast<double>(extent)
46 );
47}

49inline int32_t fromFraction(float fraction, int32_t origin, int32_t extent)
50{
51 if (extent <= 0) {
52 return origin;
53 }
54
55 const auto finiteFraction = std::isfinite(fraction) ? fraction : 0.5f;
56 const auto upper = std::nextafter(1.0f, 0.0f);
57 const auto clampedFraction = std::clamp(finiteFraction, 0.0f, upper);
58 const auto coordinate =
59 static_cast<int64_t>(origin) + static_cast<int64_t>(clampedFraction * static_cast<float>(extent));
60 return clampCoordinate(
61 static_cast<int32_t>(std::clamp(
62 coordinate, static_cast<int64_t>(std::numeric_limits<int32_t>::min()),
63 static_cast<int64_t>(std::numeric_limits<int32_t>::max())
64 )),
65 origin, extent
66 );
67}

69inline int32_t scaleDelta(int32_t delta, int speedPercent, double &remainder)
70{
71 const auto speed = std::clamp(speedPercent, kMinimumPointerSpeed, kMaximumPointerSpeed);
72 const auto scaled = static_cast<double>(delta) * static_cast<double>(speed) / 100.0 + remainder;
73 const auto integral = std::trunc(scaled);
74 if (integral >= static_cast<double>(std::numeric_limits<int32_t>::max())) {
75 remainder = 0.0;
76 return std::numeric_limits<int32_t>::max();
77 }
78 if (integral <= static_cast<double>(std::numeric_limits<int32_t>::min())) {
79 remainder = 0.0;
80 return std::numeric_limits<int32_t>::min();
81 }
82 remainder = scaled - integral;
83 return static_cast<int32_t>(integral);
84}

104inline bool
105touchesEdge(const deskflow::DisplayGeometry &display, const deskflow::DisplayGeometry &desktop, Direction edge)
106{
107 const auto displayRight = static_cast<int64_t>(display.x) + display.width;
108 const auto displayBottom = static_cast<int64_t>(display.y) + display.height;
109 const auto desktopRight = static_cast<int64_t>(desktop.x) + desktop.width;
110 const auto desktopBottom = static_cast<int64_t>(desktop.y) + desktop.height;
111
112 using enum Direction;
113 switch (edge) {
114 case Left:
115 return display.x == desktop.x;
116 case Right:
117 return displayRight == desktopRight;
118 case Top:
119 return display.y == desktop.y;
120 case Bottom:
121 return displayBottom == desktopBottom;
122 case NoDirection:
123 return false;
124 }
125 return false;
126}

138struct EdgeSpan
139{
140 int64_t begin;
141 int64_t end;
142};

144inline std::vector<EdgeSpan>
145edgeSpans(const deskflow::DisplayLayout &layout, const deskflow::DisplayGeometry &desktop, Direction edge)
146{
147 std::vector<EdgeSpan> spans;
148 spans.reserve(layout.size());
149 for (const auto &display : layout) {
150 if (display.width <= 0 || display.height <= 0 || !touchesEdge(display, desktop, edge)) {
151 continue;
152 }
153
154 const auto begin = static_cast<int64_t>(orthogonalOrigin(display, edge));
155 spans.push_back({begin, begin + static_cast<int64_t>(orthogonalExtent(display, edge))});
156 }
157
158 std::ranges::sort(spans, {}, &EdgeSpan::begin);
159 std::vector<EdgeSpan> merged;
160 merged.reserve(spans.size());
161 for (const auto &span : spans) {
162 if (merged.empty() || span.begin > merged.back().end) {
163 merged.push_back(span);
164 } else {
165 merged.back().end = std::max(merged.back().end, span.end);
166 }
167 }
168 return merged;
169}

171inline int64_t edgeLength(const std::vector<EdgeSpan> &spans)
172{
173 int64_t length = 0;
174 for (const auto &span : spans) {
175 length += span.end - span.begin;
176 }
177 return length;
178}

180inline std::optional<double> fractionAlongEdge(const std::vector<EdgeSpan> &spans, int32_t coordinate)
181{
182 const auto total = edgeLength(spans);
183 if (total <= 0) {
184 return std::nullopt;
185 }
186
187 const auto position = static_cast<int64_t>(coordinate);
188 int64_t traversed = 0;
189 for (size_t index = 0; index < spans.size(); ++index) {
190 const auto &span = spans[index];
191 const auto length = span.end - span.begin;
192 if (position >= span.begin && position < span.end) {
193 return (static_cast<double>(traversed + position - span.begin) + 0.5) / static_cast<double>(total);
194 }
195
196 if (position < span.begin) {
197 if (index == 0) {
198 return 0.5 / static_cast<double>(total);
199 }
200
201 const auto leftDistance = position - spans[index - 1].end + 1;
202 const auto rightDistance = span.begin - position;
203 const auto nearest =
204 leftDistance <= rightDistance ? static_cast<double>(traversed) - 0.5 : static_cast<double>(traversed) + 0.5;
205 return std::clamp(nearest / static_cast<double>(total), 0.0, std::nextafter(1.0, 0.0));
206 }
207 traversed += length;
208 }
209
210 return (static_cast<double>(total) - 0.5) / static_cast<double>(total);
211}

213inline std::optional<int32_t> coordinateAlongEdge(const std::vector<EdgeSpan> &spans, double fraction)
214{
215 const auto total = edgeLength(spans);
216 if (total <= 0) {
217 return std::nullopt;
218 }
219
220 const auto finiteFraction = std::isfinite(fraction) ? fraction : 0.5;
221 const auto normalized = std::clamp(finiteFraction, 0.0, std::nextafter(1.0, 0.0));
222 const auto offset = std::min(static_cast<int64_t>(normalized * static_cast<double>(total)), total - 1);
223
224 int64_t traversed = 0;
225 for (const auto &span : spans) {
226 const auto length = span.end - span.begin;
227 if (offset < traversed + length) {
228 const auto coordinate = span.begin + offset - traversed;
229 return static_cast<int32_t>(std::clamp(
230 coordinate, static_cast<int64_t>(std::numeric_limits<int32_t>::min()),
231 static_cast<int64_t>(std::numeric_limits<int32_t>::max())
232 ));
233 }
234 traversed += length;
235 }
236 return std::nullopt;
237}

239inline std::optional<int32_t> mapAcrossDisplayEdges(
240 const deskflow::DisplayLayout &sourceLayout, const deskflow::DisplayGeometry &sourceDesktop,
241 const deskflow::DisplayLayout &destinationLayout, const deskflow::DisplayGeometry &destinationDesktop,
242 Direction exitDirection, int32_t sourceCoordinate, int32_t fallbackDestinationCoordinate
243)
244{
245 static_cast<void>(fallbackDestinationCoordinate);
246 if (exitDirection == Direction::NoDirection || sourceDesktop.width <= 0 || sourceDesktop.height <= 0 ||
247 destinationDesktop.width <= 0 || destinationDesktop.height <= 0) {
248 return std::nullopt;
249 }
250
251 const auto destinationEdge = oppositeDirection(exitDirection);
252 const auto sourceSpans = edgeSpans(sourceLayout, sourceDesktop, exitDirection);
253 const auto destinationSpans = edgeSpans(destinationLayout, destinationDesktop, destinationEdge);
254 const auto fraction = fractionAlongEdge(sourceSpans, sourceCoordinate);
255 if (!fraction) {
256 return std::nullopt;
257 }
258 return coordinateAlongEdge(destinationSpans, *fraction);
259}
