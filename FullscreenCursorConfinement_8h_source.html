15class CursorConfinement
16{
17public:
18 bool owns(const Bounds &current) const
19 {
20 return m_applied && *m_applied == current;
21 }
22
23 bool owns(const Bounds &current, const Bounds &desktop) const
24 {
25 if (!m_applied) {
26 return false;
27 }
28 // Windows intersects an existing clip with the new desktop on a resolution
29 // change. It remains our restriction and must still be releasable.
30 const Bounds visible{
31 (std::max)(m_applied->left, desktop.left), (std::max)(m_applied->top, desktop.top),
32 (std::min)(m_applied->right, desktop.right), (std::min)(m_applied->bottom, desktop.bottom)
33 };
34 return owns(current) || (visible.right > visible.left && visible.bottom > visible.top && visible == current);
35 }
36
37 template <typename Apply>
38 bool update(std::optional<Bounds> target, Bounds current, const Bounds &desktop, Apply apply)
39 {
40 if (m_applied && !owns(current, desktop)) {
41 // Another app changed the clip. Its newer request must survive our release.
42 m_applied.reset();
43 m_previous.reset();
44 }
45
46 if (m_applied && (!target || *target != *m_applied)) {
47 if (!apply(m_previous)) {
48 return false;
49 }
50 current = m_previous.value_or(desktop);
51 m_applied.reset();
52 m_previous.reset();
53 }
54
55 if (!target || owns(current, desktop)) {
56 return true;
57 }
58
59 // Preserve the game's own restriction, including a one-pixel raw-input
60 // capture. On a single-display desktop no extra restriction is needed.
61 if (current.left >= target->left && current.top >= target->top && current.right <= target->right &&
62 current.bottom <= target->bottom) {
63 return true;
64 }
65
66 if (!apply(target)) {
67 return false;
68 }
69 m_previous = current == desktop ? std::nullopt : std::optional{current};
70 m_applied = target;
71 return true;
72 }
73
74private:
75 std::optional<Bounds> m_applied;
76 std::optional<Bounds> m_previous;
77};
