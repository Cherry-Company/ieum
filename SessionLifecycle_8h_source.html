21class SessionLifecycle : public QAbstractNativeEventFilter
22{
23public:
24 explicit SessionLifecycle(QGuiApplication &app) : m_app(app)
25 {
26 m_app.installNativeEventFilter(this);
27#ifdef Q_OS_WIN
28 // Background startup may never create the main widget's native window.
29 // Qt's tray HWND has its own window procedure, bypassing native filters.
30 // Keep a hidden top-level Qt HWND to receive broadcast session messages.
31 m_shutdownWindow.create();
32#endif
33 }
34
35 ~SessionLifecycle() override
36 {
37 m_app.removeNativeEventFilter(this);
38 }
39
40 bool nativeEventFilter(const QByteArray &eventType, void *message, qintptr *result) override
41 {
42 Q_UNUSED(result)
43#ifdef Q_OS_WIN
44 if (eventType == "windows_generic_MSG" || eventType == "windows_dispatcher_MSG") {
45 const auto *msg = static_cast<const MSG *>(message);
46 // WM_QUERYENDSESSION can be cancelled by another application. Windows
47 // can terminate us without another event-loop turn after WM_ENDSESSION.
48 if (msg->message == WM_ENDSESSION && msg->wParam != FALSE) {
49 diagnostic::completeSession();
50 }
51 }
52#else
53 Q_UNUSED(eventType)
54 Q_UNUSED(message)
55#endif
56 return false;
57 }
58
59private:
60 QGuiApplication &m_app;
61#ifdef Q_OS_WIN
62 QWindow m_shutdownWindow;
63#endif
64};
